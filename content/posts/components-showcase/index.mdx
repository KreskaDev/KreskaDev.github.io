## 1. Po co ten post

Każdy projekt programistyczny żyje w napięciu między abstrakcją a kodem. Diagram tłumaczy "co dlaczego", kod pokazuje "jak naprawdę". W nadchodzących postach z serii o wzorcach programistycznych — DDD, CQRS, EventStorming, mediatorzy, integracje — będę żonglował obu warstwami. Ten krótki post jest technicznym pokazem narzędzi, których będę używał.

Dla wszystkich przykładów wziąłem jeden running example: bibliotekę publiczną. Książki, czytelnicy, wypożyczenia, kary, rezerwacje — domena dostatecznie znajoma żeby nie tłumaczyć słownika, dostatecznie bogata żeby pokazać każdy wzorzec.

### 3.1 Repository Loan — C# vs TypeScript

<CodeTabs
  caption="Repository agregatu Loan — C# (EF Core) i TypeScript (Drizzle). Te same operacje, idiom typu stacka."
  snippets={[

      code: `public sealed class LoanRepository : ILoanRepository
{
    private readonly LibraryDbContext _db;

    public LoanRepository(LibraryDbContext db) => _db = db;

    public async Task<Loan?> GetById(LoanId id, CancellationToken ct)
        => await _db.Loans
            .Include(l => l.Borrower)
            .Include(l => l.Copy)
            .SingleOrDefaultAsync(l => l.Id == id, ct);

    public async Task<int> CountActiveByMember(MemberId memberId, CancellationToken ct)
        => await _db.Loans
            .CountAsync(l => l.BorrowerId == memberId && l.Status == LoanStatus.Active, ct);

    public async Task Add(Loan loan, CancellationToken ct)
    {
        await _db.Loans.AddAsync(loan, ct);
        await _db.SaveChangesAsync(ct);
    }
}`
    },
    {
      lang: 'typescript',
      filename: 'loanRepository.ts',
      code: `import { db } from './db'
import { loans } from './schema'
import { and, count, eq } from 'drizzle-orm'
import type { Loan, LoanId, MemberId } from './types'

export const loanRepository = {
  async getById(id: LoanId): Promise<Loan | null> {
    const rows = await db
      .select()
      .from(loans)
      .where(eq(loans.id, id))
      .limit(1)
    return rows[0] ?? null
  },

  async countActiveByMember(memberId: MemberId): Promise<number> {
    const rows = await db
      .select({ count: count() })
      .from(loans)
      .where(and(eq(loans.borrowerId, memberId), eq(loans.status, 'active')))
    return rows[0]?.count ?? 0
  },

  async add(loan: Loan): Promise<void> {
    await db.insert(loans).values(loan)
  },
}`
    },
  ]}

### 3.2 Python single — analiza wypożyczeń overdue

<CodeTabs
  caption="Analiza wypożyczeń przeterminowanych — Python z pandas. Naturalny idiom dla zadań data."

      lang: 'python',
      filename: 'overdue_analytics.py',
      code: `import pandas as pd
from datetime import datetime

def overdue_summary(loans_csv: str, as_of: datetime) -> pd.DataFrame:
    df = pd.read_csv(loans_csv, parse_dates=['borrowed_at', 'due_at', 'returned_at'])
    active = df[df['returned_at'].isna()].copy()
    active['overdue_days'] = (as_of - active['due_at']).dt.days
    overdue = active[active['overdue_days'] > 0]
    by_tier = (overdue
        .groupby('member_tier')
        .agg(count=('id', 'size'), avg_days=('overdue_days', 'mean'))
        .round(1))
    return by_tier.sort_values('count', ascending=False)

if __name__ == '__main__':
    summary = overdue_summary('loans.csv', datetime(2026, 6, 1))
    print(summary)
`

### 3.3 PowerShell single — seed lokalnej bazy

<CodeTabs
  caption="Skrypt seedujący lokalną bazę książkami z pliku JSON. Naturalny idiom dla dev tooling."
  snippets={[
    {
      lang: 'powershell',
      filename: 'Seed-Library.ps1',
      code: `<#
.SYNOPSIS
  Seeduje lokalną bazę book_copies z books.json + losowych shelf/condition.
#>
param(
    [Parameter(Mandatory)][string]$ConnectionString,
    [string]$BooksJsonPath = './data/books.json',
    [int]$CopiesPerBook = 3
)

$shelves = @('A1', 'A2', 'B1', 'B2', 'C1', 'C2')
$conditions = @('New', 'Good', 'Fair', 'Worn')

$books = Get-Content $BooksJsonPath | ConvertFrom-Json

foreach ($book in $books) {
    Write-Host "Seeding copies for: $($book.title)" -ForegroundColor Cyan
    1..$CopiesPerBook | ForEach-Object {
        $copyId = [guid]::NewGuid()
        $shelf = Get-Random -InputObject $shelves
        $condition = Get-Random -InputObject $conditions
        $sql = "INSERT INTO book_copies (id, book_id, shelf, condition) VALUES ('$copyId', '$($book.id)', '$shelf', '$condition');"
        Invoke-SqlCmd -ConnectionString $ConnectionString -Query $sql
    }
}
Write-Host "Done. Seeded $($books.Count * $CopiesPerBook) copies." -ForegroundColor Green
`
    },
  ]}
/>
